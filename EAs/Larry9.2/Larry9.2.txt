{ ================================================================== }
{ EA: Larry9.2 - Setup 9.2 de Larry Williams / Stormer             }
{ ================================================================== }
{ Resumo: Estrategia baseada no Setup 9.2 que opera pullbacks a     }
{ favor da tendencia definida pelo cruzamento das EMAs 9 e 21.      }
{ Apos o cruzamento (Setup 9.1), aguarda um candle de correcao      }
{ (minima menor que a anterior em alta, ou maxima maior que a       }
{ anterior em baixa). A entrada ocorre no rompimento da extrema     }
{ do candle de correcao via ordem stop. Stop loss na extrema        }
{ oposta do candle de correcao. Alvo configuravel por multiplicador }
{ do risco. Saida adicional por inversao de tendencia (EMA).        }
{ Filtros: risco minimo/maximo, horario, limite diario.             }
{ ================================================================== }


input
  { --- Periodos das medias moveis exponenciais --- }
  pPeriodoEMA9(9);                { periodo da EMA rapida }
  pPeriodoEMA21(21);              { periodo da EMA lenta }

  { --- Gatilho de entrada --- }
  pOffsetEntrada(5.0);            { deslocamento em pts alem da extrema do candle de correcao }

  { --- Gerenciamento de risco --- }
  pMultiploAlvo(1.0);             { multiplicador do risco para o alvo (1.0 = 1:1, 2.0 = 1:2, 3.0 = 1:3) }
  pRiscoMinimo(20.0);             { risco minimo em pts para aceitar o trade }
  pRiscoMaximo(300.0);            { risco maximo em pts para aceitar o trade }
  pStopOffsetPts(5.0);            { deslocamento em pts alem da extrema para o stop }

  { --- Quantidade --- }
  pQtdContratos(1);               { quantidade de mini contratos }

  { --- Controle de horario --- }
  pHoraInicio(0905);              { horario de inicio das operacoes (HHMM) }
  pHoraFim(1700);                 { horario limite para novas entradas (HHMM) }
  pHoraEncerra(1730);             { horario de encerramento forcado (HHMM) }

  { --- Gestao diaria --- }
  pMetaDiariaPts(2000.0);         { meta diaria em pontos - para de operar ao atingir }
  pLossMaxDiarioPts(1000.0);      { perda maxima diaria em pontos - para de operar ao atingir }


var
  { --- Valores das EMAs --- }
  vEMA9            : Float;
  vEMA21           : Float;
  vEMA9_ant        : Float;
  vEMA21_ant       : Float;

  { --- Tendencia --- }
  nTendencia       : Integer;     { 1 = alta, -1 = baixa, 0 = indefinido }
  nTendencia_ant   : Integer;

  { --- Candle de correcao e gatilho --- }
  bCorrecaoCompra  : Boolean;     { candle anterior eh de correcao para compra }
  bCorrecaoVenda   : Boolean;     { candle anterior eh de correcao para venda }
  vGatilhoCompra   : Float;       { preco de disparo da compra }
  vGatilhoVenda    : Float;       { preco de disparo da venda }
  vStopCompra      : Float;       { stop da compra }
  vStopVenda       : Float;       { stop da venda }
  vRiscoCompra     : Float;       { tamanho do risco da compra }
  vRiscoVenda      : Float;       { tamanho do risco da venda }
  vAlvoCompra      : Float;       { alvo da compra }
  vAlvoVenda       : Float;       { alvo da venda }
  bSetupCompraOK   : Boolean;     { setup de compra valido }
  bSetupVendaOK    : Boolean;     { setup de venda valido }

  { --- Leitura de series anteriores --- }
  vHigh1           : Float;       { maxima do candle [1] }
  vLow1            : Float;       { minima do candle [1] }
  vHigh2           : Float;       { maxima do candle [2] }
  vLow2            : Float;       { minima do candle [2] }

  { --- Posicao e gestao --- }
  vEntrada         : Float;
  vStopAtual       : Float;
  vAlvoAtual       : Float;
  nDirecao         : Integer;     { 1 = comprado, -1 = vendido }

  { --- Controle de barra --- }
  nUltimaBarra     : Integer;
  bNovaBarra       : Boolean;

  { --- Controle diario --- }
  nData            : Integer;
  bHoraOK          : Boolean;
  bLockDiario      : Boolean;
  vResultadoDia    : Float;

  { --- Controle de inversao de tendencia como saida --- }
  bInversaoEMA     : Boolean;

  { --- Auxiliar --- }
  nHoraAtual       : Integer;


begin

  { ================================================================ }
  { 1. RESET NO PRIMEIRO CANDLE                                      }
  { ================================================================ }
  if (CurrentBar = 1) then
  begin
    nData            := Date;
    nUltimaBarra     := CurrentBar;
    bNovaBarra       := False;
    bLockDiario      := False;
    nTendencia       := 0;
    nTendencia_ant   := 0;
    bCorrecaoCompra  := False;
    bCorrecaoVenda   := False;
    bSetupCompraOK   := False;
    bSetupVendaOK    := False;
    nDirecao         := 0;
    vResultadoDia    := 0;
    bInversaoEMA     := False;
  end;


  { ================================================================ }
  { 2. DETECCAO DE NOVA BARRA E NOVO DIA                             }
  { ================================================================ }
  bNovaBarra := (CurrentBar <> nUltimaBarra);
  nUltimaBarra := CurrentBar;

  { Novo dia: resetar controles diarios }
  if (Date <> nData) then
  begin
    nData         := Date;
    bLockDiario   := False;
    vResultadoDia := 0;
  end;


  { ================================================================ }
  { 3. CONTROLE DE HORARIO                                           }
  { ================================================================ }
  nHoraAtual := Time;
  bHoraOK := (nHoraAtual >= pHoraInicio) and (nHoraAtual <= pHoraFim);


  { ================================================================ }
  { 4. GESTAO DIARIA: verificar meta e loss maximo                   }
  { ================================================================ }
  vResultadoDia := DailyResult(True);
  if (vResultadoDia >= pMetaDiariaPts) or (vResultadoDia <= -pLossMaxDiarioPts) then
    bLockDiario := True;


  { ================================================================ }
  { 5. CALCULO DAS EMAs                                              }
  { ================================================================ }
  vEMA9  := MediaExp(pPeriodoEMA9, Close);
  vEMA21 := MediaExp(pPeriodoEMA21, Close);

  { Valores anteriores }
  vEMA9_ant  := MediaExp(pPeriodoEMA9, Close)[1];
  vEMA21_ant := MediaExp(pPeriodoEMA21, Close)[1];


  { ================================================================ }
  { 6. DEFINICAO DA TENDENCIA                                        }
  { ================================================================ }
  nTendencia_ant := nTendencia;

  if (vEMA9 > vEMA21) then
    nTendencia := 1
  else if (vEMA9 < vEMA21) then
    nTendencia := -1
  else
    nTendencia := 0;

  { Detectar inversao de tendencia }
  bInversaoEMA := (nTendencia <> nTendencia_ant) and (nTendencia_ant <> 0);


  { ================================================================ }
  { 7. PLOTAGEM DAS EMAs NO GRAFICO                                  }
  { ================================================================ }
  Plot(vEMA9);
  Plot2(vEMA21);


  { ================================================================ }
  { 8. SAIDAS DA POSICAO ABERTA                                      }
  { ================================================================ }

  { 8.1 Saida por inversao de tendencia das EMAs }
  if (IsBought) and (bInversaoEMA) and (nTendencia = -1) then
    ClosePosition;

  if (IsSold) and (bInversaoEMA) and (nTendencia = 1) then
    ClosePosition;

  { 8.2 Saida por horario de encerramento }
  if (HasPosition) and (nHoraAtual >= pHoraEncerra) then
    ClosePosition;

  { 8.3 Saida por lock diario (meta ou loss) }
  if (HasPosition) and (bLockDiario) then
    ClosePosition;

  { 8.4 Ordens de stop loss e take profit (cobertura) }
  if (IsBought) then
  begin
    vEntrada   := BuyPrice;
    vStopAtual := vEntrada - (vRiscoCompra);
    vAlvoAtual := vEntrada + (vRiscoCompra * pMultiploAlvo);

    { Stop de venda para protecao }
    SellToCoverStop(vStopAtual, vStopAtual);

    { Alvo limite }
    SellToCoverLimit(vAlvoAtual);
  end;

  if (IsSold) then
  begin
    vEntrada   := SellPrice;
    vStopAtual := vEntrada + (vRiscoVenda);
    vAlvoAtual := vEntrada - (vRiscoVenda * pMultiploAlvo);

    { Stop de compra para protecao }
    BuyToCoverStop(vStopAtual, vStopAtual);

    { Alvo limite }
    BuyToCoverLimit(vAlvoAtual);
  end;


  { ================================================================ }
  { 9. LEITURA DE DADOS ANTERIORES (somente em nova barra)           }
  { ================================================================ }
  if (bNovaBarra) then
  begin
    vHigh1 := High[1];
    vLow1  := Low[1];
    vHigh2 := High[2];
    vLow2  := Low[2];
  end;


  { ================================================================ }
  { 10. IDENTIFICACAO DO CANDLE DE CORRECAO                          }
  { ================================================================ }

  { --- CORRECAO PARA COMPRA ---                                     }
  { Em tendencia de alta: candle [1] com minima menor que candle [2] }
  bCorrecaoCompra := False;
  if (nTendencia = 1) and (bNovaBarra) then
  begin
    if (vLow1 < vLow2) then
      bCorrecaoCompra := True;
  end;

  { --- CORRECAO PARA VENDA ---                                      }
  { Em tendencia de baixa: candle [1] com maxima maior que candle [2]}
  bCorrecaoVenda := False;
  if (nTendencia = -1) and (bNovaBarra) then
  begin
    if (vHigh1 > vHigh2) then
      bCorrecaoVenda := True;
  end;


  { ================================================================ }
  { 11. CALCULO DOS GATILHOS, STOPS E ALVOS                          }
  { ================================================================ }

  { --- Setup 9.2 de COMPRA --- }
  bSetupCompraOK := False;
  if (bCorrecaoCompra) then
  begin
    vGatilhoCompra := vHigh1 + pOffsetEntrada;
    vStopCompra    := vLow1 - pStopOffsetPts;
    vRiscoCompra   := vGatilhoCompra - vStopCompra;
    vAlvoCompra    := vGatilhoCompra + (vRiscoCompra * pMultiploAlvo);

    { Validar filtro de risco }
    if (vRiscoCompra >= pRiscoMinimo) and (vRiscoCompra <= pRiscoMaximo) then
      bSetupCompraOK := True;
  end;

  { --- Setup 9.2 de VENDA --- }
  bSetupVendaOK := False;
  if (bCorrecaoVenda) then
  begin
    vGatilhoVenda := vLow1 - pOffsetEntrada;
    vStopVenda    := vHigh1 + pStopOffsetPts;
    vRiscoVenda   := vStopVenda - vGatilhoVenda;
    vAlvoVenda    := vGatilhoVenda - (vRiscoVenda * pMultiploAlvo);

    { Validar filtro de risco }
    if (vRiscoVenda >= pRiscoMinimo) and (vRiscoVenda <= pRiscoMaximo) then
      bSetupVendaOK := True;
  end;


  { ================================================================ }
  { 12. ENVIO DE ORDENS DE ENTRADA                                   }
  { ================================================================ }

  { Condicoes gerais para operar }
  if (not HasPosition) and (bHoraOK) and (not bLockDiario) then
  begin

    { --- Entrada COMPRA via ordem stop --- }
    if (bSetupCompraOK) then
    begin
      CancelPendingOrders;
      BuyStop(vGatilhoCompra, vGatilhoCompra, pQtdContratos);
      nDirecao := 1;
    end;

    { --- Entrada VENDA via ordem stop --- }
    if (bSetupVendaOK) then
    begin
      CancelPendingOrders;
      SellShortStop(vGatilhoVenda, vGatilhoVenda, pQtdContratos);
      nDirecao := -1;
    end;

  end;


  { ================================================================ }
  { 13. SINALIZACAO VISUAL                                           }
  { ================================================================ }

  { Pintar candle conforme posicao }
  if (IsBought) then
    PaintBar(clVerde)
  else if (IsSold) then
    PaintBar(clVermelho);

  { Marcar candle de correcao identificado }
  if (bCorrecaoCompra) and (bSetupCompraOK) and (not HasPosition) then
    PlotText("9.2C", clVerde, 0, 7);

  if (bCorrecaoVenda) and (bSetupVendaOK) and (not HasPosition) then
    PlotText("9.2V", clVermelho, 2, 7);


end;
